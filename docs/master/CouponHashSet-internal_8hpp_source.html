59template<typename A>
60uint32_t CouponHashSet<A>::getSerializedArraySize(uint8_t lgK, uint8_t lgArrInts, bool compact, uint32_t couponCount) {
61 const uint32_t maxCouponCount = hll_constants::RESIZE_NUMER * (uint32_t(1) << (lgK - 3))
62 / hll_constants::RESIZE_DENOM;
63 if (couponCount > maxCouponCount) {
64 throw std::invalid_argument("CouponHashSet count exceeds the maximum for its precision: " + std::to_string(couponCount));
65 }
66 if (lgArrInts < hll_constants::LG_INIT_SET_SIZE) {
67 lgArrInts = HllUtil<A>::computeLgArrInts(SET, couponCount, lgK);
68 }
69 if (compact) return couponCount;
70 if (lgArrInts > lgK - 3) {
71 throw std::invalid_argument("CouponHashSet table size exceeds the maximum for its precision: "
72 + std::to_string(lgArrInts));
73 }
74 const uint32_t capacity = uint32_t(1) << lgArrInts;
75 if (couponCount > hll_constants::RESIZE_NUMER * capacity / hll_constants::RESIZE_DENOM) {
76 throw std::invalid_argument("CouponHashSet count exceeds the table load limit: " + std::to_string(couponCount));
77 }
78 return capacity;
79}

81template<typename A>
82void CouponHashSet<A>::checkOccupiedSlots(const vector_int& coupons, uint32_t couponCount) {
83 const auto occupied = std::count_if(coupons.begin(), coupons.end(), [](uint32_t coupon) {
84 return coupon != hll_constants::EMPTY;
85 });
86 if (occupied != couponCount) {
87 throw std::invalid_argument("CouponHashSet count does not match the number of occupied slots: "
88 + std::to_string(couponCount) + " vs " + std::to_string(occupied));
89 }
90}

92template<typename A>
93CouponHashSet<A>* CouponHashSet<A>::newSet(const void* bytes, size_t len, const A& allocator) {
94 if (len < hll_constants::HASH_SET_INT_ARR_START) { // hard-coded
95 throw std::out_of_range("Input data length insufficient to hold CouponHashSet");
96 }
97
98 const uint8_t* data = static_cast<const uint8_t*>(bytes);
99 if (data[hll_constants::PREAMBLE_INTS_BYTE] != hll_constants::HASH_SET_PREINTS) {
100 throw std::invalid_argument("Incorrect number of preInts in input stream");
101 }
102 if (data[hll_constants::SER_VER_BYTE] != hll_constants::SER_VER) {
103 throw std::invalid_argument("Wrong ser ver in input stream");
104 }
105 if (data[hll_constants::FAMILY_BYTE] != hll_constants::FAMILY_ID) {
106 throw std::invalid_argument("Input stream is not an HLL sketch");
107 }
108
109 const hll_mode mode = HllSketchImpl<A>::extractCurMode(data[hll_constants::MODE_BYTE]);
110 if (mode != SET) {
111 throw std::invalid_argument("Calling set constructor with non-set mode data");
112 }
113
114 const target_hll_type tgtHllType = HllSketchImpl<A>::extractTgtHllType(data[hll_constants::MODE_BYTE]);
115
116 const uint8_t lgK = data[hll_constants::LG_K_BYTE];
117 if (lgK <= 7) {
118 throw std::invalid_argument("Attempt to deserialize invalid CouponHashSet with lgConfigK <= 7. Found: "
119 + std::to_string(lgK));
120 }
121 HllUtil<A>::checkLgK(lgK);
122 const uint8_t lgArrInts = data[hll_constants::LG_ARR_BYTE];
123 const bool compactFlag = ((data[hll_constants::FLAGS_BYTE] & hll_constants::COMPACT_FLAG_MASK) ? true : false);
124
125 uint32_t couponCount;
126 std::memcpy(&couponCount, data + hll_constants::HASH_SET_COUNT_INT, sizeof(couponCount));
127 const uint32_t couponsInArray = getSerializedArraySize(lgK, lgArrInts, compactFlag, couponCount);
128 const size_t arrayBytes = couponsInArray * sizeof(uint32_t);
129 const size_t expectedLength = hll_constants::HASH_SET_INT_ARR_START + arrayBytes;
130 if (len < expectedLength) {
131 throw std::out_of_range("Byte array too short for sketch. Expected " + std::to_string(expectedLength)
132 + ", found: " + std::to_string(len));
133 }
134
135 auto sketch = HllUtil<A>::template newObject<CouponHashSet<A>>(allocator, lgK, tgtHllType, allocator);
136 std::unique_ptr<CouponHashSet<A>, std::function<void(HllSketchImpl<A>*)>> ptr(sketch, sketch->get_deleter());
137
138 if (compactFlag) {
139 const uint8_t* curPos = data + hll_constants::HASH_SET_INT_ARR_START;
140 uint32_t coupon;
141 for (uint32_t i = 0; i < couponCount; ++i, curPos += sizeof(coupon)) {
142 std::memcpy(&coupon, curPos, sizeof(coupon));
143 sketch->couponUpdate(coupon);
144 }
145 } else {
146 sketch->coupons_.resize(couponsInArray);
147 std::memcpy(sketch->coupons_.data(), data + hll_constants::HASH_SET_INT_ARR_START, arrayBytes);
148 checkOccupiedSlots(sketch->coupons_, couponCount);
149 sketch->couponCount_ = couponCount;
150 }
151
152 return ptr.release();
153}

155template<typename A>
156CouponHashSet<A>* CouponHashSet<A>::newSet(std::istream& is, const A& allocator) {
157 uint8_t listHeader[8];
158 read(is, listHeader, sizeof(listHeader));
159 if (!is.good()) { throw std::runtime_error("error reading from std::istream"); }
160
161 if (listHeader[hll_constants::PREAMBLE_INTS_BYTE] != hll_constants::HASH_SET_PREINTS) {
162 throw std::invalid_argument("Incorrect number of preInts in input stream");
163 }
164 if (listHeader[hll_constants::SER_VER_BYTE] != hll_constants::SER_VER) {
165 throw std::invalid_argument("Wrong ser ver in input stream");
166 }
167 if (listHeader[hll_constants::FAMILY_BYTE] != hll_constants::FAMILY_ID) {
168 throw std::invalid_argument("Input stream is not an HLL sketch");
169 }
170
171 hll_mode mode = HllSketchImpl<A>::extractCurMode(listHeader[hll_constants::MODE_BYTE]);
172 if (mode != SET) {
173 throw std::invalid_argument("Calling set constructor with non-set mode data");
174 }
175
176 const target_hll_type tgtHllType = HllSketchImpl<A>::extractTgtHllType(listHeader[hll_constants::MODE_BYTE]);
177
178 const uint8_t lgK = listHeader[hll_constants::LG_K_BYTE];
179 if (lgK <= 7) {
180 throw std::invalid_argument("Attempt to deserialize invalid CouponHashSet with lgConfigK <= 7. Found: "
181 + std::to_string(lgK));
182 }
183 HllUtil<A>::checkLgK(lgK);
184 const uint8_t lgArrInts = listHeader[hll_constants::LG_ARR_BYTE];
185 const bool compactFlag = ((listHeader[hll_constants::FLAGS_BYTE] & hll_constants::COMPACT_FLAG_MASK) ? true : false);
186
187 uint32_t couponCount;
188 read(is, &couponCount, sizeof(couponCount));
189 if (!is.good()) { throw std::runtime_error("error reading from std::istream"); }
190 const uint32_t couponsInArray = getSerializedArraySize(lgK, lgArrInts, compactFlag, couponCount);
191
192 auto sketch = HllUtil<A>::template newObject<CouponHashSet<A>>(allocator, lgK, tgtHllType, allocator);
193 typedef std::unique_ptr<CouponHashSet<A>, std::function<void(HllSketchImpl<A>*)>> coupon_hash_set_ptr;
194 coupon_hash_set_ptr ptr(sketch, sketch->get_deleter());
195
196 if (compactFlag) {
197 for (uint32_t i = 0; i < couponCount; ++i) {
198 uint32_t coupon;
199 read(is, &coupon, sizeof(coupon));
200 if (!is.good()) { throw std::runtime_error("error reading from std::istream"); }
201 sketch->couponUpdate(coupon);
202 }
203 } else {
204 sketch->coupons_.resize(couponsInArray);
205 read(is, sketch->coupons_.data(), couponsInArray * sizeof(uint32_t));
206 if (!is.good()) { throw std::runtime_error("error reading from std::istream"); }
207 checkOccupiedSlots(sketch->coupons_, couponCount);
208 sketch->couponCount_ = couponCount;
209 }
210
211 return ptr.release();
212}

224template<typename A>
225HllSketchImpl<A>* CouponHashSet<A>::couponUpdate(uint32_t coupon) {
226 const uint8_t lgCouponArrInts = count_trailing_zeros_in_u32(static_cast<uint32_t>(this->coupons_.size()));
227 const int32_t index = find<A>(this->coupons_.data(), lgCouponArrInts, coupon);
228 if (index >= 0) {
229 return this; // found duplicate, ignore
230 }
231 this->coupons_[~index] = coupon; // found empty
232 ++this->couponCount_;
233 if (static_cast<size_t>(hll_constants::RESIZE_DENOM * this->couponCount_) > (hll_constants::RESIZE_NUMER * this->coupons_.size())) {
234 return growOrPromoteWithRollback(static_cast<uint32_t>(~index));
235 }
236 return this;
237}

249template<typename A>
250HllSketchImpl<A>* CouponHashSet<A>::growOrPromoteWithRollback(uint32_t index) {
251 try {
252 const uint8_t lgCouponArrInts = count_trailing_zeros_in_u32(static_cast<uint32_t>(this->coupons_.size()));
253 if (lgCouponArrInts == (this->lgConfigK_ - 3)) { // at max size
254 return this->promoteHeapListOrSetToHll(*this);
255 }
256 growHashSet(static_cast<uint8_t>(lgCouponArrInts + 1));
257 } catch (...) {
258 this->coupons_[index] = hll_constants::EMPTY;
259 --this->couponCount_;
260 throw;
261 }
262 return this;
263}

265template<typename A>
266void CouponHashSet<A>::growHashSet(uint8_t tgtLgCoupArrSize) {
267 const uint32_t tgtLen = 1 << tgtLgCoupArrSize;
268 vector_int coupons_new(tgtLen, 0, this->coupons_.get_allocator());
269
270 const uint32_t srcLen = static_cast<uint32_t>(this->coupons_.size());
271 for (uint32_t i = 0; i < srcLen; ++i) { // scan existing array for non-zero values
272 const uint32_t fetched = this->coupons_[i];
273 if (fetched != hll_constants::EMPTY) {
274 const int32_t idx = find<A>(coupons_new.data(), tgtLgCoupArrSize, fetched); // search TGT array
275 if (idx < 0) { // found EMPTY
276 coupons_new[~idx] = fetched; // insert
277 continue;
278 }
279 throw std::runtime_error("Error: Found duplicate coupon");
280 }
281 }
282 this->coupons_ = std::move(coupons_new);
283}

285template<typename A>
286static int32_t find(const uint32_t* array, uint8_t lgArrInts, uint32_t coupon) {
287 const uint32_t arrMask = (1 << lgArrInts) - 1;
288 uint32_t probe = coupon & arrMask;
289 const uint32_t loopIndex = probe;
290 do {
291 const uint32_t couponAtIdx = array[probe];
292 if (couponAtIdx == hll_constants::EMPTY) {
293 return ~probe; //empty
294 }
295 else if (coupon == couponAtIdx) {
296 return probe; //duplicate
297 }
298 const uint32_t stride = ((coupon & hll_constants::KEY_MASK_26) >> lgArrInts) | 1;
299 probe = (probe + stride) & arrMask;
300 } while (probe != loopIndex);
301 throw std::invalid_argument("Key not found and no empty slots!");
302}
