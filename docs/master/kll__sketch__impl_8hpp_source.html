189template<typename T, typename C, typename A>
190template<typename FwdT>
191void kll_sketch<T, C, A>::update(FwdT&& item) {
192 if (!check_update_item(item)) { return; }
193 check_n_increase(1);
194 allocate_items(); // first, so that a failure leaves the sketch unchanged
195 reset_sorted_view(); // before any change, so that a failed copy can't leave a stale view
196 update_min_max(static_cast<const T&>(item)); // min and max are always copies
197 insert_level_zero(std::forward<FwdT>(item));
198}

200template<typename T, typename C, typename A>
201template<typename FwdT>
202void kll_sketch<T, C, A>::insert_level_zero(FwdT&& item) {
203 if (levels_[0] == 0) {
204 // item can refer into items_, and internal_update() can destroy it
205 T copy(std::forward<FwdT>(item));
206 const uint32_t index = internal_update();
207 new (&items_[index]) T(std::move(copy));
208 } else {
209 // construct the item before counting it, so that a failed copy leaves the sketch unchanged
210 new (&items_[levels_[0] - 1]) T(std::forward<FwdT>(item));
211 internal_update();
212 }
213}

222template<typename T, typename C, typename A>
223template<typename FwdT>
224void kll_sketch<T, C, A>::update(FwdT&& item, uint64_t weight) {
225 if (!check_update_item(item)) { return; }
226 if (weight == 0) throw std::invalid_argument("weight must be at least 1");
227 if (weight > kll_constants::MAX_WEIGHT) {
228 throw std::invalid_argument("weight must be less than 2^61: " + std::to_string(weight));
229 }
230 check_n_increase(weight);
231 if (weight == 1) {
232 update(std::forward<FwdT>(item));
233 return;
234 }
235 allocate_items(); // first, so that a failure leaves the sketch unchanged
236 reset_sorted_view(); // before any change, so that a failed copy can't leave a stale view
237 if (weight < levels_[0]) { // fits into level zero without compaction
238 update_min_max(static_cast<const T&>(item));
239 // each copy is counted only once it is constructed, so a failed copy leaves a valid sketch
240 for (uint64_t i = 1; i < weight; ++i) insert_level_zero(static_cast<const T&>(item));
241 insert_level_zero(std::forward<FwdT>(item));
242 } else {
243 merge(kll_sketch(k_, static_cast<const T&>(item), weight, comparator_, allocator_));
244 }
245}

247// An item at level h carries weight 2^h, so one copy of the item at each level
248// whose bit is set in the weight is an exact sketch of that many copies.
249// The weight is less than 2^61, so this needs at most 61 levels.
250template<typename T, typename C, typename A>
251kll_sketch<T, C, A>::kll_sketch(uint16_t k, const T& item, uint64_t weight, const C& comparator, const A& allocator):
252comparator_(comparator),
253allocator_(allocator),
254k_(k),
255m_(kll_constants::DEFAULT_M),
256min_k_(k),
257num_levels_(static_cast<uint8_t>(64 - count_leading_zeros_in_u64(weight))),
258is_level_zero_sorted_(true),
259n_(weight),
260levels_(num_levels_ + 1, 0, allocator),
261items_(nullptr),
262items_size_(0),
263min_item_(item),
264max_item_(item),
265sorted_view_(nullptr)
266{
267 for (uint8_t level = 0; level < num_levels_; ++level) {
268 levels_[level + 1] = levels_[level] + static_cast<uint32_t>((weight >> level) & 1);
269 }
270 items_size_ = levels_[num_levels_];
271 items_ = allocator_.allocate(items_size_);
272 // the destructor doesn't run if a constructor throws, so release the items here
273 uint32_t num_constructed = 0;
274 try {
275 for (; num_constructed < items_size_; ++num_constructed) new (&items_[num_constructed]) T(item);
276 } catch (...) {
277 for (uint32_t i = 0; i < num_constructed; ++i) items_[i].~T();
278 allocator_.deallocate(items_, items_size_);
279 throw;
280 }
281}

283template<typename T, typename C, typename A>
284void kll_sketch<T, C, A>::update_min_max(const T& item) {
285 if (is_empty()) {
286 min_item_.emplace(item);
287 max_item_.emplace(item);
288 } else {
289 if (comparator_(item, *min_item_)) *min_item_ = item;
290 if (comparator_(*max_item_, item)) *max_item_ = item;
291 }

294template<typename T, typename C, typename A>
295void kll_sketch<T, C, A>::reserve() {
296 if (items_ == nullptr) {
297 // nothing is stored yet, so the buffer can start at its full size
298 items_ = allocator_.allocate(k_);
299 items_size_ = k_;
300 levels_[0] = levels_[1] = k_;
301 } else if (num_levels_ == 1 && items_size_ < k_) {
302 grow_items(k_); // the buffer is still growing: grow it to its full size in one step
303 reset_sorted_view(); // the cached view points into the old buffer
304 }
305}

312template<typename T, typename C, typename A>
313template<typename O>
314void kll_sketch<T, C, A>::copy_items(const O& other) {
315 if (other.items_ == nullptr) return;
316 allocate_items();
317 auto i = levels_[0];
318 try {
319 for (; i < levels_[num_levels_]; ++i) new (&items_[i]) T(other.items_[i]);
320 } catch (...) {
321 // the destructor doesn't run if a constructor throws, so release the items here
322 for (auto j = levels_[0]; j < i; ++j) items_[j].~T();
323 allocator_.deallocate(items_, items_size_);
324 items_ = nullptr;
325 throw;
326 }
327}

334template<typename T, typename C, typename A>
335uint32_t kll_sketch<T, C, A>::internal_update() {
336 allocate_items();
337 if (levels_[0] == 0) {
338 if (items_size_ < k_) {
339 grow_items(std::min<uint32_t>(items_size_ << static_cast<uint8_t>(RESIZE_FACTOR), k_));
340 } else {
341 compress_while_updating();
342 }
343 }
344 n_++;
345 is_level_zero_sorted_ = false;
346 return --levels_[0];
347}

349template<typename T, typename C, typename A>
350template<typename FwdSk>
351void kll_sketch<T, C, A>::merge(FwdSk&& other) {
352 if (this == &other) { throw std::invalid_argument("A sketch cannot merge with itself."); }
353 if (other.is_empty()) { return; }
354 if (m_ != other.m_) {
355 throw std::invalid_argument("incompatible M: " + std::to_string(m_) + " and " + std::to_string(other.m_));
356 }
357 check_n_increase(other.n_);
358 allocate_items(); // the higher levels are merged into items_ even if other's level zero is empty
359 reset_sorted_view(); // before any change, so that a failed copy can't leave a stale view
360 if (is_empty()) {
361 min_item_.emplace(conditional_forward<FwdSk>(*other.min_item_));
362 max_item_.emplace(conditional_forward<FwdSk>(*other.max_item_));
363 } else {
364 if (comparator_(*other.min_item_, *min_item_)) *min_item_ = conditional_forward<FwdSk>(*other.min_item_);
365 if (comparator_(*max_item_, *other.max_item_)) *max_item_ = conditional_forward<FwdSk>(*other.max_item_);
366 }
367 const uint64_t final_n = n_ + other.n_;
368 for (uint32_t i = other.levels_[0]; i < other.levels_[1]; i++) {
369 insert_level_zero(conditional_forward<FwdSk>(other.items_[i]));
370 }
371 if (other.num_levels_ >= 2) { merge_higher_levels(std::forward<FwdSk>(other), final_n); }
372 n_ = final_n;
373 if (other.is_estimation_mode()) { min_k_ = std::min(min_k_, other.min_k_); }
374 assert_correct_total_weight();
375}

464size_t kll_sketch<T, C, A>::get_serialized_size_bytes(const SerDe&) const {
465 if (is_empty()) { return EMPTY_SIZE_BYTES; }
466 if (num_levels_ == 1 && get_num_retained() == 1) {
467 return DATA_START_SINGLE_ITEM + sizeof(TT);
468 }
469 // the last integer in the levels_ array is not serialized because it can be derived
470 return DATA_START + num_levels_ * sizeof(uint32_t) + (get_num_retained() + 2) * sizeof(TT);

475template<typename TT, typename SerDe, typename std::enable_if<!std::is_arithmetic<TT>::value, int>::type>
476size_t kll_sketch<T, C, A>::get_serialized_size_bytes(const SerDe& sd) const {
477 if (is_empty()) { return EMPTY_SIZE_BYTES; }
478 if (num_levels_ == 1 && get_num_retained() == 1) {
479 return DATA_START_SINGLE_ITEM + sd.size_of_item(items_[levels_[0]]);
480 }
481 // the last integer in the levels_ array is not serialized because it can be derived
482 size_t size = DATA_START + num_levels_ * sizeof(uint32_t);
483 size += sd.size_of_item(*min_item_);
484 size += sd.size_of_item(*max_item_);
485 for (auto it: *this) size += sd.size_of_item(it.first);
486 return size;
487}

509template<typename T, typename C, typename A>
510template<typename SerDe>
511void kll_sketch<T, C, A>::serialize(std::ostream& os, const SerDe& sd) const {
512 const bool is_single_item = n_ == 1;
513 const uint8_t preamble_ints(is_empty() || is_single_item ? PREAMBLE_INTS_SHORT : PREAMBLE_INTS_FULL);
514 write(os, preamble_ints);
515 const uint8_t serial_version(is_single_item ? SERIAL_VERSION_2 : SERIAL_VERSION_1);
516 write(os, serial_version);
517 const uint8_t family(FAMILY);
518 write(os, family);
519 const uint8_t flags_byte(
520 (is_empty() ? 1 << flags::IS_EMPTY : 0)
521 | (is_level_zero_sorted_ ? 1 << flags::IS_LEVEL_ZERO_SORTED : 0)
522 | (is_single_item ? 1 << flags::IS_SINGLE_ITEM : 0)
523 );
524 write(os, flags_byte);
525 write(os, k_);
526 write(os, m_);
527 const uint8_t unused = 0;
528 write(os, unused);
529 if (is_empty()) return;
530 if (!is_single_item) {
531 write(os, n_);
532 write(os, min_k_);
533 write(os, num_levels_);
534 write(os, unused);
535 if (num_levels_ == 1 && items_size_ < k_) {
536 write(os, levels_[0] + (k_ - items_size_));
537 } else {
538 write(os, levels_.data(), sizeof(levels_[0]) * num_levels_);
539 }
540 sd.serialize(os, &*min_item_, 1);
541 sd.serialize(os, &*max_item_, 1);
542 }
543 sd.serialize(os, &items_[levels_[0]], get_num_retained());
544}

546template<typename T, typename C, typename A>
547template<typename SerDe>
548auto kll_sketch<T, C, A>::serialize(unsigned header_size_bytes, const SerDe& sd) const -> vector_bytes {
549 const bool is_single_item = n_ == 1;
550 const size_t size = header_size_bytes + get_serialized_size_bytes(sd);
551 vector_bytes bytes(size, 0, allocator_);
552 uint8_t* ptr = bytes.data() + header_size_bytes;
553 const uint8_t* end_ptr = bytes.data() + size;
554 const uint8_t preamble_ints(is_empty() || is_single_item ? PREAMBLE_INTS_SHORT : PREAMBLE_INTS_FULL);
555 ptr += copy_to_mem(preamble_ints, ptr);
556 const uint8_t serial_version(is_single_item ? SERIAL_VERSION_2 : SERIAL_VERSION_1);
557 ptr += copy_to_mem(serial_version, ptr);
558 const uint8_t family(FAMILY);
559 ptr += copy_to_mem(family, ptr);
560 const uint8_t flags_byte(
561 (is_empty() ? 1 << flags::IS_EMPTY : 0)
562 | (is_level_zero_sorted_ ? 1 << flags::IS_LEVEL_ZERO_SORTED : 0)
563 | (is_single_item ? 1 << flags::IS_SINGLE_ITEM : 0)
564 );
565 ptr += copy_to_mem(flags_byte, ptr);
566 ptr += copy_to_mem(k_, ptr);
567 ptr += copy_to_mem(m_, ptr);
568 ptr += sizeof(uint8_t); // unused
569 if (!is_empty()) {
570 if (!is_single_item) {
571 ptr += copy_to_mem(n_, ptr);
572 ptr += copy_to_mem(min_k_, ptr);
573 ptr += copy_to_mem(num_levels_, ptr);
574 ptr += sizeof(uint8_t); // unused
575 if (num_levels_ == 1 && items_size_ < k_) {
576 ptr += copy_to_mem(levels_[0] + (k_ - items_size_), ptr);
577 } else {
578 ptr += copy_to_mem(levels_.data(), ptr, sizeof(levels_[0]) * num_levels_);
579 }
580 ptr += sd.serialize(ptr, end_ptr - ptr, &*min_item_, 1);
581 ptr += sd.serialize(ptr, end_ptr - ptr, &*max_item_, 1);
582 }
583 const size_t bytes_remaining = end_ptr - ptr;
584 ptr += sd.serialize(ptr, bytes_remaining, &items_[levels_[0]], get_num_retained());
585 }
586 const size_t delta = ptr - bytes.data();
587 if (delta != size) throw std::logic_error("serialized size mismatch: " + std::to_string(delta)
588 + " != " + std::to_string(size));
589 return bytes;
590}

592template<typename T, typename C, typename A>
593template<typename SerDe>
594kll_sketch<T, C, A> kll_sketch<T, C, A>::deserialize(std::istream& is, const SerDe& sd,
595 const C& comparator, const A& allocator) {
596 const auto preamble_ints = read<uint8_t>(is);
597 const auto serial_version = read<uint8_t>(is);
598 const auto family_id = read<uint8_t>(is);
599 const auto flags_byte = read<uint8_t>(is);
600 const auto k = read<uint16_t>(is);
601 const auto m = read<uint8_t>(is);
602 read<uint8_t>(is); // skip unused byte
603
604 check_k(k);
605 check_m(m);
606 check_preamble_ints(preamble_ints, flags_byte);
607 check_serial_version(serial_version);
608 check_family_id(family_id);
609
610 if (!is.good()) throw std::runtime_error("error reading from std::istream");
611 const bool is_empty(flags_byte & (1 << flags::IS_EMPTY));
612 if (is_empty) return kll_sketch(k, comparator, allocator);
613
614 uint64_t n;
615 uint16_t min_k;
616 uint8_t num_levels;
617 const bool is_single_item(flags_byte & (1 << flags::IS_SINGLE_ITEM)); // used in serial version 2
618 if (is_single_item) {
619 n = 1;
620 min_k = k;
621 num_levels = 1;
622 } else {
623 n = read<uint64_t>(is);
624 min_k = read<uint16_t>(is);
625 num_levels = read<uint8_t>(is);
626 read<uint8_t>(is); // skip unused byte
627 check_num_levels(num_levels);
628 }
629 vector_u32 levels(num_levels + 1, 0, allocator);
630 const uint32_t capacity(kll_helper::compute_total_capacity(k, m, num_levels));
631 if (is_single_item) {
632 levels[0] = capacity - 1;
633 } else {
634 // the last integer in levels_ is not serialized because it can be derived
635 read(is, levels.data(), sizeof(levels[0]) * num_levels);
636 if (!is.good()) throw std::runtime_error("error reading from std::istream");
637 }
638 levels[num_levels] = capacity;
639 check_levels(levels);
640 optional<T> min_item;
641 optional<T> max_item;
642 if (!is_single_item) {
643 // Space to deserialize min and max.
644 // serde::deserialize expects allocated but not initialized storage.
645 typename std::aligned_storage<sizeof(T), alignof(T)>::type tmp_storage;
646 T* tmp = reinterpret_cast<T*>(&tmp_storage);
647
648 sd.deserialize(is, tmp, 1);
649 // serde call did not throw, repackage and cleanup
650 min_item.emplace(std::move(*tmp));
651 tmp->~T();
652 sd.deserialize(is, tmp, 1);
653 // serde call did not throw, repackage and cleanup
654 max_item.emplace(std::move(*tmp));
655 tmp->~T();
656 }
657 A alloc(allocator);
658 auto items_buffer_deleter = [capacity, &alloc](T* ptr) { alloc.deallocate(ptr, capacity); };
659 std::unique_ptr<T, decltype(items_buffer_deleter)> items_buffer(alloc.allocate(capacity), items_buffer_deleter);
660 const auto num_items = levels[num_levels] - levels[0];
661 sd.deserialize(is, &items_buffer.get()[levels[0]], num_items);
662 // serde call did not throw, repackage with destrtuctors
663 std::unique_ptr<T, items_deleter> items(items_buffer.release(), items_deleter(levels[0], capacity, allocator));
664 const bool is_level_zero_sorted = (flags_byte & (1 << flags::IS_LEVEL_ZERO_SORTED)) > 0;
665 if (is_single_item) {
666 min_item.emplace(items.get()[levels[0]]);
667 max_item.emplace(items.get()[levels[0]]);
668 }
669 if (!is.good())
670 throw std::runtime_error("error reading from std::istream");
671 return kll_sketch(k, min_k, n, num_levels, std::move(levels), std::move(items), capacity,
672 std::move(min_item), std::move(max_item), is_level_zero_sorted, comparator);
673}

802// The following code is only valid in the special case of exactly reaching capacity while updating.
803// It cannot be used while merging, while reducing k, or anything else.
804template<typename T, typename C, typename A>
805void kll_sketch<T, C, A>::compress_while_updating(void) {
806 const uint8_t level = find_level_to_compact();
807
808 // It is important to add the new top level right here. Be aware that this operation
809 // grows the buffer and shifts the data and also the boundaries of the data and grows the
810 // levels array and increments num_levels_
811 if (level == (num_levels_ - 1)) {
812 add_empty_top_level_to_completely_full_sketch();
813 }
814
815 const uint32_t raw_beg = levels_[level];
816 const uint32_t raw_lim = levels_[level + 1];
817 // +2 is OK because we already added a new top level if necessary
818 const uint32_t pop_above = levels_[level + 2] - raw_lim;
819 const uint32_t raw_pop = raw_lim - raw_beg;
820 const bool odd_pop = kll_helper::is_odd(raw_pop);
821 const uint32_t adj_beg = odd_pop ? raw_beg + 1 : raw_beg;
822 const uint32_t adj_pop = odd_pop ? raw_pop - 1 : raw_pop;
823 const uint32_t half_adj_pop = adj_pop / 2;
824 const uint32_t destroy_beg = levels_[0];
825
826 // level zero might not be sorted, so we must sort it if we wish to compact it
827 // sort_level_zero() is not used here because of the adjustment for odd number of items
828 if ((level == 0) && !is_level_zero_sorted_) {
829 std::sort(items_ + adj_beg, items_ + adj_beg + adj_pop, comparator_);
830 }
831 if (pop_above == 0) {
832 kll_helper::randomly_halve_up(items_, adj_beg, adj_pop);
833 } else {
834 kll_helper::randomly_halve_down(items_, adj_beg, adj_pop);
835 kll_helper::merge_sorted_arrays<T, C>(items_, adj_beg, half_adj_pop, raw_lim, pop_above, adj_beg + half_adj_pop, comparator_);
836 }
837 levels_[level + 1] -= half_adj_pop; // adjust boundaries of the level above
838 if (odd_pop) {
839 levels_[level] = levels_[level + 1] - 1; // the current level now contains one item
840 if (levels_[level] != raw_beg) items_[levels_[level]] = std::move(items_[raw_beg]); // namely this leftover guy
841 } else {
842 levels_[level] = levels_[level + 1]; // the current level is now empty
843 }
844
845 // verify that we freed up half_adj_pop array slots just below the current level
846 if (levels_[level] != (raw_beg + half_adj_pop)) throw std::logic_error("compaction error");
847
848 // finally, we need to shift up the data in the levels below
849 // so that the freed-up space can be used by level zero
850 if (level > 0) {
851 const uint32_t amount = raw_beg - levels_[0];
852 std::move_backward(items_ + levels_[0], items_ + levels_[0] + amount, items_ + levels_[0] + half_adj_pop + amount);
853 for (uint8_t lvl = 0; lvl < level; lvl++) levels_[lvl] += half_adj_pop;
854 }
855 for (uint32_t i = 0; i < half_adj_pop; i++) items_[i + destroy_beg].~T();
856}

858template<typename T, typename C, typename A>
859uint8_t kll_sketch<T, C, A>::find_level_to_compact() const {
860 uint8_t level = 0;
861 while (true) {
862 if (level >= num_levels_) throw std::logic_error("capacity calculation error");
863 const uint32_t pop = levels_[level + 1] - levels_[level];
864 const uint32_t cap = kll_helper::level_capacity(k_, num_levels_, level, m_);
865 if (pop >= cap) {
866 return level;
867 }
868 level++;
869 }
870}

872template<typename T, typename C, typename A>
873void kll_sketch<T, C, A>::add_empty_top_level_to_completely_full_sketch() {
874 const uint32_t cur_total_cap = levels_[num_levels_];
875
876 // make sure that we are following a certain growth scheme
877 if (levels_[0] != 0) throw std::logic_error("full sketch expected");
878 if (items_size_ != cur_total_cap) throw std::logic_error("current capacity mismatch");
879
880 // note that merging MIGHT over-grow levels_, in which case we might not have to grow it here
881 const uint8_t new_levels_size = num_levels_ + 2;
882 if (levels_.size() < new_levels_size) {
883 levels_.resize(new_levels_size);
884 }
885
886 const uint32_t delta_cap = kll_helper::level_capacity(k_, num_levels_ + 1, 0, m_);
887 const uint32_t new_total_cap = cur_total_cap + delta_cap;
888
889 // move (and shift) the current data into the new buffer
890 T* new_buf = allocator_.allocate(new_total_cap);
891 kll_helper::move_construct<T>(items_, 0, cur_total_cap, new_buf, delta_cap, true);
892 allocator_.deallocate(items_, items_size_);
893 items_ = new_buf;
894 items_size_ = new_total_cap;
895
896 // this loop includes the old "extra" index at the top
897 for (uint8_t i = 0; i <= num_levels_; i++) {
898 levels_[i] += delta_cap;
899 }
900
901 if (levels_[num_levels_] != new_total_cap) throw std::logic_error("new capacity mismatch");
902
903 num_levels_++;
904 levels_[num_levels_] = new_total_cap; // initialize the new "extra" index at the top
905}

907// Grows the items buffer of a one-level sketch when it is full and smaller than k.
908// Compaction starts only when the buffer is full at k items, so growing does not change the result.
909template<typename T, typename C, typename A>
910void kll_sketch<T, C, A>::grow_items(uint32_t new_size) {
911 const uint32_t delta = new_size - items_size_;
912 T* new_items = allocator_.allocate(new_size);
913 kll_helper::move_construct<T>(items_, levels_[0], levels_[1], new_items, levels_[0] + delta, true);
914 allocator_.deallocate(items_, items_size_);
915 items_ = new_items;
916 items_size_ = new_size;
917 levels_[0] += delta;
918 levels_[1] += delta;
919}

956template<typename T, typename C, typename A>
957template<typename O>
958void kll_sketch<T, C, A>::merge_higher_levels(O&& other, uint64_t final_n) {
959 const uint32_t tmp_num_items = get_num_retained() + other.get_num_retained_above_level_zero();
960 A alloc(allocator_);
961 auto tmp_items_deleter = [tmp_num_items, &alloc](T* ptr) { alloc.deallocate(ptr, tmp_num_items); }; // no destructor needed
962 const std::unique_ptr<T, decltype(tmp_items_deleter)> workbuf(allocator_.allocate(tmp_num_items), tmp_items_deleter);
963 const uint8_t ub = kll_helper::ub_on_num_levels(final_n);
964 const size_t work_levels_size = ub + 2; // ub+1 does not work
965 vector_u32 worklevels(work_levels_size, 0, allocator_);
966 vector_u32 outlevels(work_levels_size, 0, allocator_);
967
968 const uint8_t provisional_num_levels = std::max(num_levels_, other.num_levels_);
969
970 // Allocate everything the merge needs before any item is moved out of this sketch,
971 // so that an allocation failure leaves this sketch valid.
972 // The level sizes after compression depend only on the level sizes before it.
973 worklevels[0] = 0;
974 worklevels[1] = safe_level_size(0); // the level zero items of other were already inserted into this
975 for (uint8_t lvl = 1; lvl < provisional_num_levels; lvl++) {
976 worklevels[lvl + 1] = worklevels[lvl] + safe_level_size(lvl) + other.safe_level_size(lvl);
977 }
978 const kll_helper::compress_result predicted = kll_helper::predict_compress(k_, m_, provisional_num_levels,
979 worklevels.data(), outlevels.data());
980 // ub can sometimes be much bigger
981 if (predicted.final_num_levels > ub) throw std::logic_error("merge error");
982 auto new_items_deleter = [&predicted, &alloc](T* ptr) { alloc.deallocate(ptr, predicted.final_capacity); }; // holds no items
983 std::unique_ptr<T, decltype(new_items_deleter)> new_items(
984 predicted.final_capacity != items_size_ ? allocator_.allocate(predicted.final_capacity) : nullptr, new_items_deleter);
985 const size_t new_levels_size = predicted.final_num_levels + 1;
986 levels_.reserve(new_levels_size);
987
988 populate_work_arrays(std::forward<O>(other), workbuf.get(), worklevels.data(), provisional_num_levels);
989
990 const kll_helper::compress_result result = kll_helper::general_compress<T, C>(k_, m_, provisional_num_levels, workbuf.get(),
991 worklevels.data(), outlevels.data(), is_level_zero_sorted_, comparator_);
992 if (result.final_num_levels != predicted.final_num_levels || result.final_capacity != predicted.final_capacity
993 || result.final_num_items != predicted.final_num_items) {
994 throw std::logic_error("merge error: compression does not match the prediction");
995 }
996
997 // now we need to transfer the results back into "this" sketch
998 if (new_items) {
999 allocator_.deallocate(items_, items_size_);
1000 items_size_ = result.final_capacity;
1001 items_ = new_items.release();
1002 }
1003 const uint32_t free_space_at_bottom = result.final_capacity - result.final_num_items;
1004 kll_helper::move_construct<T>(workbuf.get(), outlevels[0], outlevels[0] + result.final_num_items, items_, free_space_at_bottom, true);
1005
1006 if (levels_.size() < new_levels_size) {
1007 levels_.resize(new_levels_size); // within the reserved capacity
1008 }
1009 const uint32_t offset = free_space_at_bottom - outlevels[0];
1010 for (uint8_t lvl = 0; lvl < levels_.size(); lvl++) { // includes the "extra" index
1011 levels_[lvl] = outlevels[lvl] + offset;
1012 }
1013 num_levels_ = result.final_num_levels;
1014}
1015
1016// This leaves items_ uninitialized (all objects moved out and destroyed).
1017// worklevels must already hold the level boundaries in workbuf.
1018// The items of other are copied (or moved) first, the only step that can fail on a throwing copy:
1019// if one throws, the copies made are destroyed and this sketch is unchanged.
1020// After that, the items of this sketch are only moved.
1021template<typename T, typename C, typename A>
1022template<typename FwdSk>
1023void kll_sketch<T, C, A>::populate_work_arrays(FwdSk&& other, T* workbuf, const uint32_t* worklevels, uint8_t provisional_num_levels) {
1024 // the level zero items of other were already inserted into this
1025 // each level of other goes after the items of this sketch at that level
1026 uint8_t lvl = 1;
1027 uint32_t num_constructed = 0; // at level lvl
1028 try {
1029 for (; lvl < provisional_num_levels; ++lvl) {
1030 const uint32_t other_pop = other.safe_level_size(lvl);
1031 const uint32_t dst = worklevels[lvl] + safe_level_size(lvl);
1032 for (num_constructed = 0; num_constructed < other_pop; ++num_constructed) {
1033 new (&workbuf[dst + num_constructed]) T(conditional_forward<FwdSk>(other.items_[other.levels_[lvl] + num_constructed]));
1034 }
1035 }
1036 } catch (...) {
1037 for (uint8_t l = 1; l <= lvl && l < provisional_num_levels; ++l) {
1038 const uint32_t dst = worklevels[l] + safe_level_size(l);
1039 const uint32_t count = l < lvl ? other.safe_level_size(l) : num_constructed;
1040 for (uint32_t i = 0; i < count; ++i) workbuf[dst + i].~T();
1041 }
1042 throw;
1043 }
1044
1045 kll_helper::move_construct<T>(items_, levels_[0], levels_[1], workbuf, 0, true);
