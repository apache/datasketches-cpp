29template<typename T, typename C, typename A>
30quantiles_sorted_view<T, C, A>::quantiles_sorted_view(uint32_t num, const C& comparator, const A& allocator):
31comparator_(comparator),
32total_weight_(0),
33min_added_(false),
34entries_(allocator)
35{
36 if (num > 0) // the if exists to optimise for empty sketches
37 entries_.reserve(num + 2); // room for the min and max
38}

40template<typename T, typename C, typename A>
41template<typename Iterator>
42void quantiles_sorted_view<T, C, A>::add(Iterator first, Iterator last, uint64_t weight) {
43 if (first == last) return;
44 const size_t size_before = entries_.size();
45 for (auto it = first; it != last; ++it) entries_.emplace_back(ref_helper(*it), weight);
46 if (size_before > (min_added_ ? 1 : 0)) { // an added min is smaller than all items, so it needs no merge
47 Container tmp(entries_.get_allocator());
48 tmp.reserve(entries_.capacity());
49 std::merge(
50 entries_.begin(), entries_.begin() + size_before,
51 entries_.begin() + size_before, entries_.end(),
52 std::back_inserter(tmp), compare_pairs_by_first(comparator_)
53 );
54 std::swap(tmp, entries_);
55 }
56}

58template<typename T, typename C, typename A>
59void quantiles_sorted_view<T, C, A>::convert_to_cummulative() {
60 for (auto& entry: entries_) {
61 total_weight_ += entry.second;
62 entry.second = total_weight_;
63 }
64}

66template<typename T, typename C, typename A>
67void quantiles_sorted_view<T, C, A>::add_min(const T& min_item) {
68 entries_.emplace_back(ref_helper(min_item), 1);
69 min_added_ = true;
70}

72template<typename T, typename C, typename A>
73void quantiles_sorted_view<T, C, A>::convert_to_cummulative(const T& max_item) {
74 if (min_added_) --entries_[1].second;
75 if (comparator_(deref_helper(entries_.back().first), max_item)) {
76 --entries_.back().second;
77 entries_.push_back(Entry(ref_helper(max_item), 1));
78 }
79 convert_to_cummulative();
80}

82template<typename T, typename C, typename A>
83double quantiles_sorted_view<T, C, A>::get_rank(const T& item, bool inclusive) const {
84 if (entries_.empty()) throw std::runtime_error("operation is undefined for an empty sketch");
85 auto it = inclusive ?
86 std::upper_bound(entries_.begin(), entries_.end(), Entry(ref_helper(item), 0), compare_pairs_by_first(comparator_))
87 : std::lower_bound(entries_.begin(), entries_.end(), Entry(ref_helper(item), 0), compare_pairs_by_first(comparator_));
88 // we need item just before
89 if (it == entries_.begin()) return 0;
90 --it;
91 return static_cast<double>(it->second) / total_weight_;
92}

94template<typename T, typename C, typename A>
95auto quantiles_sorted_view<T, C, A>::get_quantile(double rank, bool inclusive) const -> quantile_return_type {
96 if (entries_.empty()) throw std::runtime_error("operation is undefined for an empty sketch");
97 uint64_t weight = static_cast<uint64_t>(inclusive ? std::ceil(rank * total_weight_) : rank * total_weight_);
98 auto it = inclusive ?
99 std::lower_bound(entries_.begin(), entries_.end(), make_dummy_entry<T>(weight), compare_pairs_by_second())
100 : std::upper_bound(entries_.begin(), entries_.end(), make_dummy_entry<T>(weight), compare_pairs_by_second());
101 if (it == entries_.end()) return deref_helper(entries_[entries_.size() - 1].first);
102 return deref_helper(it->first);
103}

105template<typename T, typename C, typename A>
106auto quantiles_sorted_view<T, C, A>::get_CDF(const T* split_points, uint32_t size, bool inclusive) const -> vector_double {
107 if (entries_.empty()) throw std::runtime_error("operation is undefined for an empty sketch");
108 check_split_points(split_points, size);
109 vector_double ranks(entries_.get_allocator());
110 ranks.reserve(size + 1);
111 for (uint32_t i = 0; i < size; ++i) ranks.push_back(get_rank(split_points[i], inclusive));
112 ranks.push_back(1);
113 return ranks;
114}

116template<typename T, typename C, typename A>
117auto quantiles_sorted_view<T, C, A>::get_PMF(const T* split_points, uint32_t size, bool inclusive) const -> vector_double {
118 auto buckets = get_CDF(split_points, size, inclusive);
119 for (uint32_t i = size; i > 0; --i) {
120 buckets[i] -= buckets[i - 1];
121 }
122 return buckets;
123}
