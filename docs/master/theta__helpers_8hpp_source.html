52template<typename ExtractKey, typename Entry, typename Allocator>
53static uint64_t trim_to_nominal(std::vector<Entry, Allocator>& entries, uint32_t nominal_size, uint64_t theta) {
54 if (entries.size() <= nominal_size) return theta;
55 std::nth_element(entries.begin(), entries.begin() + nominal_size, entries.end(), compare_by_key<ExtractKey>());
56 const uint64_t new_theta = ExtractKey()(entries[nominal_size]);
57 entries.erase(entries.begin() + nominal_size, entries.end());
58 entries.shrink_to_fit();
59 return new_theta;
60}

96 static uint8_t starting_sub_multiple(uint8_t lg_tgt, uint8_t lg_min, uint8_t lg_rf) {
97 return (lg_tgt <= lg_min) ? lg_min : (lg_rf == 0) ? lg_tgt : ((lg_tgt - lg_min) % lg_rf) + lg_min;
