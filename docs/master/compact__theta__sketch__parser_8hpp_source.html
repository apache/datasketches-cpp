80 // The MASK selects which bits of the first 8 bytes of an empty image are examined.
81 // The TEST gives the required values of the examined bits; it must lie within the MASK.
82 // Examined: preamble longs, serial version 3, sketch type 3, bytes 3 and 4 = 0 (not used by compact);
83 // flags read-only, empty, compact set; single-item and reserved bits 0, 6, 7 clear.
84 // The ordered flag is ignored: C++ before 3.3.0 and Java before 1.0.0 wrote empty images without it.
85 // The reserved flag bits must be zero for serial version 3. Any future use of them requires a new one.
86 // The seed hash is ignored in the 8-byte form: it may be 0 or the seed hash, depending on the writer.
87 // These must stay the same as in Java EmptyCompactSketch.
88 static const uint64_t EMPTY_SKETCH_MASK = 0x0000EFFFFFFFFFFFULL;
89 static const uint64_t EMPTY_SKETCH_TEST = 0x00000E0000030301ULL;
90 static const uint64_t EMPTY_SKETCH_TEST_LEGACY = 0x00000E0000030303ULL;
91 static_assert((EMPTY_SKETCH_TEST & ~EMPTY_SKETCH_MASK) == 0, "EMPTY_SKETCH_TEST must lie within EMPTY_SKETCH_MASK");
92 static_assert((EMPTY_SKETCH_TEST_LEGACY & ~EMPTY_SKETCH_MASK) == 0, "EMPTY_SKETCH_TEST_LEGACY must lie within EMPTY_SKETCH_MASK");
